@interact(year=widgets.IntSlider(
    min=df['year'].min(), max=df['year'].max(), step=1, value=df['year'].min()))
def show_map(year):
    filtered = df[df['year'] == year].copy()

    # Create display column for tooltip with NaN handling
    filtered["pct_diff_display"] = filtered["Growth variability"].apply(
        lambda x: "non-bearing" if pd.isna(x) else f"{x:.2f}"
    )

    # Ensure GeoDataFrame with CRS
    if not isinstance(filtered, gpd.GeoDataFrame):
        filtered = gpd.GeoDataFrame(filtered, geometry='geometry')
    if filtered.crs is None:
        filtered.set_crs("EPSG:4326", inplace=True)

    # Fix centroid warning using projected CRS
    projected = filtered.to_crs(epsg=3857)
    centroids = projected.geometry.centroid
    lat = centroids.to_crs(epsg=4326).y.max()
    lon = centroids.to_crs(epsg=4326).x.max()

    # Define red → yellow → green ramp
    colormap = cm.LinearColormap(
        colors=["red", "yellow", "green"],
        vmin=df["Growth variability"].min(),
        vmax=df["Growth variability"].max()
    )
    colormap.caption = "Growth variability"

    # Style function for color mapping and NaNs
    def style_function(feature):
        value = feature['properties']['Growth variability']
        if value is None or pd.isna(value):
            return {
                'fillColor': 'none',
                'color': 'black',
                'weight': 1,
                'fillOpacity': 0
            }
        else:
            return {
                'fillColor': colormap(value),
                'color': 'black',
                'weight': 1,
                'fillOpacity': 0.6
            }

    # Tooltip using preprocessed display column
    tooltip = folium.GeoJsonTooltip(
        fields=["pct_diff_display"],
        aliases=["Growth variability"],
        localize=True,
        sticky=True,
        labels=True,
        style=("background-color: white; color: #333; font-family: arial; font-size: 12px; padding: 5px;")
    )

    # Add GeoJson to map
    folium.Map(location=[lat, lon], zoom_start=13, tiles="CartoDB positron")
    m = folium.Map(location=[lat, lon], zoom_start=13, tiles="CartoDB positron")

    folium.GeoJson(
        filtered,
        name="fields",
        style_function=style_function,
        tooltip=tooltip
    ).add_to(m)

    colormap.add_to(m)
    folium.LayerControl().add_to(m)
    display(m)

def create_map_for_year(year, save_html=False):
    filtered = df[df['year'] == year].copy()

    # Ensure it's a valid GeoDataFrame with CRS
    if not isinstance(filtered, gpd.GeoDataFrame):
        filtered = gpd.GeoDataFrame(filtered, geometry='geometry')
    if filtered.crs is None:
        filtered.set_crs("EPSG:4326", inplace=True)

    # Calculate centroids
    centroids = filtered.geometry.centroid
    lat = centroids.y.mean()
    lon = centroids.x.mean()

    # Set up colormap
    values = filtered['pct_diff_from_P95_avg']
    colormap = cm.linear.YlOrRd_09.scale(values.min(), values.max())
    colormap.caption = "Pct diff from P95 avg"

    # Create folium map
    m = folium.Map(location=[lat, lon], zoom_start=13, tiles="CartoDB positron")

    folium.GeoJson(
        filtered,
        name="fields",
        style_function=lambda feature: {
            'fillColor': colormap(feature['properties']['pct_diff_from_P95_avg']),
            'color': 'black',
            'weight': 1,
            'fillOpacity': 0.6
        },
        tooltip=folium.GeoJsonTooltip(fields=["pct_diff_from_P95_avg"])
    ).add_to(m)

    colormap.add_to(m)
    folium.LayerControl().add_to(m)

    if save_html:
        m.save(f"map_{year}.html")
        print(f"Saved map to map_{year}.html")

    return m
